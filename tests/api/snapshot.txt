== dxfio/boxed.hpp
#include <memory>
#include <utility>
namespace DxfIO
{
  template <class ValueT>
  class Boxed
  {
  public:
    Boxed() : m_Value(std::make_unique<ValueT>()) {}
    explicit Boxed(ValueT value) : m_Value(std::make_unique<ValueT>(std::move(value))) {}
    Boxed(const Boxed &other) : m_Value(Copied(other)) {}
    Boxed(Boxed &&other) noexcept = default;
    auto operator=(const Boxed &other) -> Boxed &
    {
      if (this == &other)
      {
        return *this;
      }
      if (m_Value && other.m_Value)
      {
        *m_Value = *other.m_Value;
        return *this;
      }
      m_Value = Copied(other);
      return *this;
    }
    auto operator=(Boxed &&other) noexcept -> Boxed & = default;
    ~Boxed() = default;
    [[nodiscard]] auto Valueless() const noexcept -> bool { return !m_Value; }
    auto operator*() -> ValueT & { return Held(); }
    auto operator*() const -> const ValueT & { return m_Value ? *m_Value : Default(); }
    auto operator->() -> ValueT * { return &Held(); }
    auto operator->() const -> const ValueT * { return &**this; }
    auto operator==(const Boxed &other) const -> bool { return **this == *other; }
  private:
    static auto Copied(const Boxed &other) -> std::unique_ptr<ValueT>
    {
      return other.m_Value ? std::make_unique<ValueT>(*other.m_Value) : nullptr;
    }
    static auto Default() -> const ValueT &
    {
      static const ValueT empty{};
      return empty;
    }
    auto Held() -> ValueT &
    {
      if (!m_Value)
      {
        m_Value = std::make_unique<ValueT>();
      }
      return *m_Value;
    }
    std::unique_ptr<ValueT> m_Value;
  };
  template <class ValueT>
  auto Unboxed(ValueT &value) -> ValueT &
  {
    return value;
  }
  template <class ValueT>
  auto Unboxed(const ValueT &value) -> const ValueT &
  {
    return value;
  }
  template <class ValueT>
  auto Unboxed(Boxed<ValueT> &value) -> ValueT &
  {
    return *value;
  }
  template <class ValueT>
  auto Unboxed(const Boxed<ValueT> &value) -> const ValueT &
  {
    return *value;
  }
}

    requires std::is_enum_v<FlagT>
  [[nodiscard]] constexpr auto Has(int flags, FlagT flag) -> bool

{
  auto operator()(const DxfIO::Symbol &one) const noexcept -> std::size_t
  {
    return std::hash<const void *>{}(&one.Str());
  }
};

    return std::formatter<std::string_view>::format(one.View(), context);
  }
};

{
  [[nodiscard]] DXFIO_API auto Matches(std::string_view one, std::string_view other) -> bool;
  [[nodiscard]] DXFIO_API auto Block(Drawing &drawing, std::string_view name)
      -> Blocks::Definition *;
  [[nodiscard]] DXFIO_API auto Block(const Drawing &drawing, std::string_view name)
      -> const Blocks::Definition *;
  auto Block(const Drawing &&drawing, std::string_view name) -> const Blocks::Definition * = delete;
  template <class RecordT>
    requires requires(const RecordT &record) { record.Record.Name; }
  [[nodiscard]] auto Record(Tables::Table<RecordT> &table, std::string_view name) -> RecordT *
  {
    const auto found = std::ranges::find_if(table.Records, [name](const RecordT &record) -> bool
                                            { return Matches(record.Record.Name, name); });
    return found == table.Records.end() ? nullptr : &*found;
  }
  template <class RecordT>
    requires requires(const RecordT &record) { record.Record.Name; }
  [[nodiscard]] auto Record(const Tables::Table<RecordT> &table, std::string_view name)
      -> const RecordT *
  {
    const auto found = std::ranges::find_if(table.Records, [name](const RecordT &record) -> bool
                                            { return Matches(record.Record.Name, name); });
    return found == table.Records.end() ? nullptr : &*found;
  }
  template <class RecordT, class DrawingT>
    requires std::same_as<std::remove_const_t<DrawingT>, Drawing>
  [[nodiscard]] auto TableOf(DrawingT &drawing) -> auto &
  {
    if constexpr (std::same_as<RecordT, Tables::VPort>)
    {
      return drawing.VPort;
    }
    else if constexpr (std::same_as<RecordT, Tables::LType>)
    {
      return drawing.LType;
    }
    else if constexpr (std::same_as<RecordT, Tables::Layer>)
    {
      return drawing.Layer;
    }
    else if constexpr (std::same_as<RecordT, Tables::Style>)
    {
      return drawing.Style;
    }
    else if constexpr (std::same_as<RecordT, Tables::View>)
    {
      return drawing.View;
    }
    else if constexpr (std::same_as<RecordT, Tables::Ucs>)
    {
      return drawing.Ucs;
    }
    else if constexpr (std::same_as<RecordT, Tables::AppId>)
    {
      return drawing.AppId;
    }
    else if constexpr (std::same_as<RecordT, Tables::DimStyle>)
    {
      return drawing.DimStyle;
    }
    else
    {
      static_assert(std::same_as<RecordT, Tables::BlockRecord>, "a kind of table record");
      return drawing.BlockRecord;
    }
  }
  template <class RecordT>
  [[nodiscard]] auto Record(Drawing &drawing, std::string_view name) -> RecordT *
  {
    return Record(TableOf<RecordT>(drawing), name);
  }
  template <class RecordT>
  [[nodiscard]] auto Record(const Drawing &drawing, std::string_view name) -> const RecordT *
  {
    return Record(TableOf<RecordT>(drawing), name);
  }
  template <class RecordT>
  auto Record(const Drawing &&drawing, std::string_view name) -> const RecordT * = delete;
